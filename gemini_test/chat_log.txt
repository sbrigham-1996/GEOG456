// Define the region of interest (Santa Cruz, Bolivia)
var roi = ee.Geometry.Rectangle([-64, -16, -59, -19]);

// Define the dry season (June to August)
var startDate = '1985-06-01'; // Landsat 5 start date
var endDate = '1985-08-31';

// Filter Landsat 5 collection for the dry season and region
var l5Collection = ee.ImageCollection('LANDSAT/LT05/C02/T1_L2')
  .filterDate(startDate, endDate)
  .filterBounds(roi);

// Function to mask clouds and cloud shadows
function maskL8sr(image) {
  // Bit 0 is cloud, bit 3 is cloud shadow
  var cloudShadowBitMask = (1 << 3);
  var cloudsBitMask = (1 << 5); // Dilated Cloud
  var qa = image.select('QA_PIXEL');

  // Both flags should be set to zero, indicating clear conditions.
  var mask = qa.bitwiseAnd(cloudShadowBitMask).eq(0)
                 .and(qa.bitwiseAnd(cloudsBitMask).eq(0));
  return image.updateMask(mask);
}

// Apply cloud mask to each image in the collection
var maskedL5Collection = l5Collection.map(maskL8sr);


// Function to calculate NDVI
var calculateNDVI = function(image) {
  // Surface reflectance bands are renamed in Landsat 5 Collection 2
  var nir = image.select('SR_B4'); 
  var red = image.select('SR_B3');

  var ndvi = nir.subtract(red).divide(nir.add(red)).rename('NDVI');
  return image.addBands(ndvi);
};

// Calculate NDVI for each image in the masked collection
var ndviCollection = maskedL5Collection.map(calculateNDVI);

// Calculate the median NDVI for the dry season
var medianNDVI = ndviCollection.median().clip(roi).select('NDVI');

// Define visualization parameters for NDVI
var ndviVis = {
  min: -1,
  max: 1,
  palette: ['blue', 'white', 'green']
};

// Display the median NDVI image
Map.centerObject(roi, 9);
Map.addLayer(medianNDVI, ndviVis, 'Median NDVI');


// Optionally export the median NDVI image to Google Drive or Asset
Export.image.toDrive({
  image: medianNDVI,
  description: 'Median_NDVI_SantaCruz_DrySeason_1985',
  scale: 30, // Set the desired scale
  region: roi,
  maxPixels: 1e13 // Important for large areas
});


// Print some collection info to the console

print('Number of images after cloud masking:', maskedL5Collection.size());


// Add the original Landsat image as a layer for visual comparison (Optional)

    * **Input Raster:** Select your image.
    * **Method:**  Choose the appropriate method for your imagery.  "Multispectral" works well for Sentinel-2 and some Landsat.
    * **Output Raster:** Specify the output location.

3. **Run the Tool:** Click **Run**.

**Method 3: Manual Masking with Classification Tools (Most Time-Consuming, More Control):**

1. **Add Data:** Add your imagery to ArcGIS Pro.

2. **Create Training Samples:** Use the **Image Classification** toolbar to create training samples for "cloud" and "no cloud" areas.  Carefully select representative samples.

3. **Classify the Image:** Use a classification method like **Maximum Likelihood Classification** or **Support Vector Machine** to create a cloud mask.

4. **Refine the Mask (Optional):**  Use raster editing tools to clean up any misclassified areas.


**Important Considerations:**

* **Image Preprocessing:** Atmospheric correction is highly recommended before cloud masking and NDVI calculation to minimize atmospheric effects that might be confused with clouds.
* **Shadow Masking:**  Consider also masking out cloud shadows, as they can also affect NDVI.
* **Accuracy Assessment:**  Visually inspect your cloud masks to ensure accuracy.


By applying these methods, you can create cloud-free imagery for your NDVI time series analysis, leading to more reliable results.  Remember to choose the method that best suits your data and resources.  The function chain method using quality assessment bands is generally preferred for its accuracy and automation capabilities when available.
